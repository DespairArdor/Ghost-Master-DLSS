cmake_minimum_required(VERSION 3.24)
project(GhostMasterDLSS VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ghostmaster_probe SHARED src/probe.cpp)
target_include_directories(ghostmaster_probe PRIVATE external/reshade/include)
target_compile_definitions(ghostmaster_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ghostmaster_probe PRIVATE /W4 /EHsc)
target_link_libraries(ghostmaster_probe PRIVATE d3d12 dxguid)
set_target_properties(ghostmaster_probe PROPERTIES SUFFIX ".addon64")
add_executable(gpu_probe src/gpu_probe.cpp)
target_link_libraries(gpu_probe PRIVATE d3d12 dxgi)
target_compile_definitions(gpu_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(dlss_backend STATIC src/dlss_backend.cpp)
target_include_directories(dlss_backend PUBLIC external/DLSS/include src)
target_compile_definitions(dlss_backend PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(dlss_backend PUBLIC "${CMAKE_SOURCE_DIR}/external/DLSS/lib/Windows_x86_64/x64/nvsdk_ngx_d.lib")
add_executable(ngx_smoke src/ngx_smoke.cpp)
target_link_libraries(ngx_smoke PRIVATE dlss_backend d3d12 dxgi)
add_executable(stack_probe src/stack_probe.cpp)
target_link_libraries(stack_probe PRIVATE dbghelp)
add_library(ghostmaster_dlss SHARED src/probe.cpp src/live_bridge.cpp src/version.rc)
target_include_directories(ghostmaster_dlss PRIVATE external/reshade/include external/reshade/deps/imgui)
target_compile_definitions(ghostmaster_dlss PRIVATE GM_LIVE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ghostmaster_dlss PRIVATE dlss_backend ghostmaster_fg d3d12 dxguid d3dcompiler wintrust crypt32)
set_target_properties(ghostmaster_dlss PROPERTIES SUFFIX ".addon64")
add_executable(bridge_smoke src/ngx_smoke.cpp src/live_bridge.cpp)
target_compile_definitions(bridge_smoke PRIVATE GM_BRIDGE_SMOKE)
target_link_libraries(bridge_smoke PRIVATE dlss_backend d3d12 dxgi d3dcompiler)
add_library(ghostmaster_fg SHARED src/frame_generation.cpp src/version.rc)
target_include_directories(ghostmaster_fg PRIVATE external/Streamline/include)
target_compile_definitions(ghostmaster_fg PRIVATE GM_FG_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ghostmaster_fg PRIVATE d3d12 dxgi wintrust crypt32)
add_executable(sl_smoke src/sl_smoke.cpp)
target_compile_definitions(sl_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(sl_smoke PRIVATE ghostmaster_fg dlss_backend d3d12 dxgi d3dcompiler)
